cmake_minimum_required(VERSION 3.22)
project(loemby_mpv_jni LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT ANDROID_ABI STREQUAL "arm64-v8a")
    message(FATAL_ERROR "Android mpv currently requires arm64-v8a")
endif()
if(NOT MPV_NATIVE_DIR)
    message(FATAL_ERROR "MPV_NATIVE_DIR missing; run scripts/android-mpv.py prepare")
endif()
add_library(mpv SHARED IMPORTED)
set_target_properties(mpv PROPERTIES IMPORTED_LOCATION "${MPV_NATIVE_DIR}/jniLibs/${ANDROID_ABI}/libmpv.so")
add_library(avcodec SHARED IMPORTED)
set_target_properties(avcodec PROPERTIES IMPORTED_LOCATION "${MPV_NATIVE_DIR}/jniLibs/${ANDROID_ABI}/libavcodec.so")
add_library(loemby_mpv_jni SHARED mpv_jni.cpp)
target_include_directories(loemby_mpv_jni PRIVATE "${MPV_NATIVE_DIR}/include")
target_compile_options(loemby_mpv_jni PRIVATE -Wall -Wextra)
target_link_options(loemby_mpv_jni PRIVATE -Wl,-z,max-page-size=16384 -Wl,-z,common-page-size=16384)
target_link_libraries(loemby_mpv_jni PRIVATE mpv avcodec android log)
